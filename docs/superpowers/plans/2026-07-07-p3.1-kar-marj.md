```js
function bnsKarMarj(b) {
  var m = (typeof b.maliyet === 'number') ? b.maliyet : null;
  var s = (typeof b.satis === 'number') ? b.satis : null;
  if (m == null && s == null) return { kar: null, marj: null };
  var kar = (s || 0) - (m || 0);
  var marj = (s && s > 0) ? Math.round((kar / s) * 100) : null;
  return { kar: kar, marj: marj };
}
function bnsFinansOzet(briefs) {
  var satis = 0, maliyet = 0, kar = 0, faturalanmamis = 0, tahsilEdilmemis = 0;
  (briefs || []).forEach(function (b) {
    var km = bnsKarMarj(b);
    if (km.kar != null) kar += km.kar;
    if (typeof b.satis === 'number') { satis += b.satis; if (!b.fatura) faturalanmamis += b.satis; else if (!b.odeme) tahsilEdilmemis += b.satis; }
    if (typeof b.maliyet === 'number') maliyet += b.maliyet;
  });
  var marj = satis > 0 ? Math.round((kar / satis) * 100) : null;
  return { satis: satis, maliyet: maliyet, kar: kar, marj: marj, faturalanmamis: faturalanmamis, tahsilEdilmemis: tahsilEdilmemis };
}
```
`module.exports = { ... , bnsKarMarj, bnsFinansOzet }` (mevcut export listesine ekle).

### Task 4: Ody `finans_ozet` tool'u

**Files:** Modify `server/ody-tools.js`

- [ ] **Step 1: calc require + tool** — dosya başına `const calc = require('../dashboard/app/calc');` (formula-test aynı yolu kullanıyor). `const TOOLS`'tan önce:
```js
defs.finans_ozet = {
  description: "Kârlılık özeti — YALNIZ yönetici. kapsam: firma (genel kâr/marj/faturalanmamış/tahsil) | marka (bir markanın kârı; marka gerekir). Kâr=satış−maliyet, tamamlanan işlerden (seçili aralık).",
  input_schema: { type: 'object', required: ['kapsam'], properties: { kapsam: { type: 'string', enum: ['firma','marka'] }, marka: { type: 'string' } } },
  run: async (input, ctx) => {
    // Yönetici kapısı: admin VEYA USERS kaydında rol='yonetici'
    const me = (ctx.ed.bns_users || []).find(u => u && u.id === (ctx.user && ctx.user.slack_id));
    const yonetici = (ctx.isAdmin) || (me && me.rol === 'yonetici');
    if (!yonetici) return { hata: 'bu bilgi yöneticilere özel' };
    let done = ctx.ed.bns_completed || [];
    if (input.kapsam === 'marka') {
      const q = String(input.marka || '').toLocaleLowerCase('tr');
      done = done.filter(c => (c.marka || '').toLocaleLowerCase('tr').includes(q));
      if (!done.length) return { hata: 'marka bulunamadı / tamamlanan iş yok' };
    }
    const o = calc.bnsFinansOzet(done);
    return { kapsam: input.kapsam, marka: input.marka || null, ...o };
  },
};
```
NOT: `ctx.ed.bns_completed`'in finans alanlarını içerdiğini teyit et (SEC-4 sonrası tam veri geliyor). İçermezse getEmbedded çağrısı sensitive olmalı — ama Ody server-side (x-bns-token / kendi çağrısı) tam veri alır; teyit et.
